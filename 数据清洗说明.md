## 5. 清洗代码（`preprocess.py`）

```python
#AI辅助说明：DeepSeek-V4-Pro，v4，DeepSeek，2026-07-29，
#生成/辅助内容：完整数据清洗流水线代码结构、缺失值处理逻辑、
#缩尾处理实现、分类变量编码实现、任务三标签构造。
#该内容经作者核对修改后用于 B 题数据预处理及论文支撑材料。

import pandas as pd
import numpy as np
import json
from pathlib import Path
from sklearn.preprocessing import LabelEncoder

# ============================================================
# 1. Load raw data
# ============================================================
root = Path(r'D:\Documents\建模')
csv_path = list(root.rglob('*B题数据集.csv'))[0]
df = pd.read_csv(csv_path, encoding='utf-8')
print(f"Original: {df.shape}")

# ============================================================
# 2. Missing indicators (BEFORE filling)
# ============================================================
df['education_level_missing'] = df['education_level'].isna().astype(int)
df['charging_station_accessibili_ms'] = df['charging_station_accessibility'].isna().astype(int)
df['ev_knowledge_score_ms'] = df['ev_knowledge_score'].isna().astype(int)

# ============================================================
# 3. education_level: NaN -> 'Missing' string (per blueprint)
# ============================================================
df['education_level'] = df['education_level'].fillna('Missing')

# ============================================================
# 4. fuel_expense outlier: <0 -> NaN, flag
# ============================================================
df['fuel_expense_invalid'] = (df['fuel_expense_per_month'] < 0).astype(int)
df.loc[df['fuel_expense_per_month'] < 0, 'fuel_expense_per_month'] = np.nan

# ============================================================
# 5. Fill numeric NaN with median
# ============================================================
num_cols_with_na = ['charging_station_accessibility',
                    'ev_knowledge_score', 'fuel_expense_per_month']
for col in num_cols_with_na:
    med = df[col].median()
    df[col] = df[col].fillna(med)

# ============================================================
# 6. Winsorization (1%/99% for continuous variables)
# ============================================================
winsor_cols = [
    'age', 'annual_income', 'daily_commute_km',
    'weekly_travel_distance_km', 'vehicle_age_years',
    'fuel_expense_per_month', 'charging_station_accessibility',
    'nearest_charging_station_km', 'electricity_cost_per_kwh',
    'environmental_awareness_score', 'government_incentive_awareness',
    'technology_affinity_score', 'range_anxiety_score',
    'battery_replacement_concern', 'ev_knowledge_score',
    'monthly_energy_consumption_kwh', 'monthly_charging_cost'
]

winsor_bounds = {}
for col in winsor_cols:
    lo = df[col].quantile(0.01)
    hi = df[col].quantile(0.99)
    winsor_bounds[col] = {'lower': float(lo), 'upper': float(hi)}
    df[col] = df[col].clip(lo, hi)

with open('winsor_bounds.json', 'w', encoding='utf-8') as f:
    json.dump(winsor_bounds, f, indent=2, ensure_ascii=False)

# ============================================================
# 7. Derived numeric features (after winsorization)
# ============================================================
df['ln_fuel'] = np.log(df['fuel_expense_per_month'].clip(lower=1e-10))
df['ln_income'] = np.log(df['annual_income'])

edu_map_ord = {'High School': 1, 'Bachelor': 2, 'Master': 3, 'PhD': 4}
df['edu_ord'] = df['education_level'].map(edu_map_ord)
edu_ord_med = df.loc[df['edu_ord'].notna(), 'edu_ord'].median()
df['edu_ord'] = df['edu_ord'].fillna(edu_ord_med)

df['annual_income_ms'] = 0
df['daily_commute_km_ms'] = 0

target_map = {'Low': 1, 'Medium': 2, 'High': 3}
df['target_ord'] = df['ev_adoption_likelihood'].map(target_map)

# ============================================================
# 8. Label-encode categorical variables
# ============================================================
cat_cols = ['education_level', 'city_type', 'current_vehicle_type']
encoding_maps = {}

for col in cat_cols:
    le = LabelEncoder()
    df[col] = le.fit_transform(df[col])
    mapping = {label: int(idx) for idx, label in enumerate(le.classes_)}
    encoding_maps[col] = mapping

with open('encoding_maps.json', 'w', encoding='utf-8') as f:
    json.dump(encoding_maps, f, indent=2, ensure_ascii=False)

# ============================================================
# 9. Task 3 label
# ============================================================
df['task3_anxiety_label'] = (df['range_anxiety_score'] > 5).astype(int)

# ============================================================
# 10. Save
# ============================================================
df.to_excel('processed_data.xlsx', index=False)
print(f"Done. Shape: {df.shape}, No NaN: {df.isnull().sum().sum() == 0}")
```
